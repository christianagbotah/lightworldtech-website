    62→function useTypewriter(messages: string[], speed = 45) {
    63→  const [text, setText] = useState('');
    64→  const [msgIdx, setMsgIdx] = useState(0);
    65→  const [deleting, setDeleting] = useState(false);
    66→
    67→  useEffect(() => {
    68→    const current = messages[msgIdx];
    69→    const timeout = setTimeout(() => {
    70→      if (!deleting) {
    71→        const next = current.slice(0, text.length + 1);
    72→        setText(next);
    73→        if (next.length === current.length) {
    74→          setTimeout(() => setDeleting(true), 2800);
    75→        }
    76→      } else {
    77→        const next = text.slice(0, -1);
    78→        setText(next);
    79→        if (next.length === 0) {
    80→          setDeleting(false);
    81→          setMsgIdx((prev) => (prev + 1) % messages.length);
    82→        }
    83→      }
    84→    }, deleting ? 22 : speed);
    85→    return () => clearTimeout(timeout);
    86→  }, [text, msgIdx, deleting, messages, speed]);
    87→
    88→  return text;
    89→}

    91→function useCountUp(end: number, duration = 2000, decimals = 0, delay = 600) {
    92→  const [count, setCount] = useState(0);
    93→  const [started, setStarted] = useState(false);
    94→
    95→  useEffect(() => {
    96→    const t = setTimeout(() => setStarted(true), delay);
    97→    return () => clearTimeout(t);
    98→  }, [delay]);
    99→
   100→  useEffect(() => {
   101→    if (!started) return;
   102→    const start = Date.now();
   103→    const interval = setInterval(() => {
   104→      const progress = Math.min((Date.now() - start) / duration, 1);
   105→      const eased = 1 - Math.pow(1 - progress, 3);
   106→      setCount(Number((eased * end).toFixed(decimals)));
   107→      if (progress >= 1) clearInterval(interval);
   108→    }, 16);
   109→    return () => clearInterval(interval);
   110→  }, [started, end, duration, decimals]);
   111→
   112→  return count;
   113→}

   119→/** Deterministic seeded PRNG (mulberry32) — keeps SSR & client output identical. */
   120→function mulberry32(seed: number) {
   121→  return function () {
   122→    let t = (seed += 0x6d2b79f5);
   123→    t = Math.imul(t ^ (t >>> 15), t | 1);
   124→    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
   125→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
   126→  };
   127→}

   816→};
   817→
   818→export default function HomePage() {
   819→  const [currentSlide, setCurrentSlide] = useState(0);
   820→  const [isTransitioning, setIsTransitioning] = useState(false);
   821→  const [isPaused, setIsPaused] = useState(false);
   822→  const touchStartY = useRef(0);
   823→  const containerRef = useRef<HTMLDivElement>(null);
   824→  const lastAutoTime = useRef(Date.now());
   825→
   826→  useSEO({
   827→    title: 'Lightworld Technologies — Powering Digital Innovation',
   828→    description: 'Leading IT solutions provider in Ghana. Web development, cloud services, cybersecurity, data analytics, and enterprise software.',
   829→    keywords: ['IT company Ghana', 'Lightworld Technologies', 'web development Ghana', 'cybersecurity Africa', 'cloud services'],
   830→  });
   831→
   832→  // Auto-advance every 6 seconds
   833→  useEffect(() => {
   834→    if (isPaused) return;
   835→    const interval = setInterval(() => {
   836→      if (!isPaused && !isTransitioning) {
   837→        setCurrentSlide((prev) => (prev + 1) % SLIDES.length);
   838→      }
   839→    }, 7000);
   840→    return () => clearInterval(interval);
   841→  }, [isPaused, isTransitioning]);
   842→
   843→  const goToSlide = useCallback((index: number) => {
   844→    if (isTransitioning || index === currentSlide) return;
   845→    setIsTransitioning(true);
   846→    setCurrentSlide(index);
   847→    setTimeout(() => setIsTransitioning(false), 700);
   848→  }, [currentSlide, isTransitioning]);
   849→
   850→  const nextSlide = useCallback(() => goToSlide((currentSlide + 1) % SLIDES.length), [currentSlide, goToSlide]);
   851→  const prevSlide = useCallback(() => goToSlide((currentSlide - 1 + SLIDES.length) % SLIDES.length), [currentSlide, goToSlide]);
   852→
   853→  // Wheel handler
   854→  useEffect(() => {
   855→    let lastWheel = 0;
   856→    const handleWheel = (e: WheelEvent) => {
   857→      e.preventDefault();
   858→      const now = Date.now();
   859→      if (now - lastWheel < 1000) return; // Throttle
   860→      lastWheel = now;
   861→
   862→      if (Math.abs(e.deltaY) > 30) {
   863→        if (e.deltaY > 0) nextSlide();
   864→        else prevSlide();
   865→      }
   866→    };
   867→
   868→    const el = containerRef.current;
   869→    if (el) {
   870→      el.addEventListener('wheel', handleWheel, { passive: false });
   871→      return () => el.removeEventListener('wheel', handleWheel);
   872→    }
   873→  }, [nextSlide, prevSlide]);
   874→
   875→  // Touch handlers
   876→  const handleTouchStart = useCallback((e: React.TouchEvent) => {
   877→    touchStartY.current = e.touches[0].clientY;
   878→  }, []);
   879→
   880→  const handleTouchEnd = useCallback((e: React.TouchEvent) => {
   881→    const diff = touchStartY.current - e.changedTouches[0].clientY;
   882→    if (Math.abs(diff) > 50) {
   883→      if (diff > 0) nextSlide();
   884→      else prevSlide();
   885→    }
   886→  }, [nextSlide, prevSlide]);
   887→
   888→  // Keyboard
   889→  useEffect(() => {
   890→    const handleKey = (e: KeyboardEvent) => {
   891→      if (e.key === 'ArrowDown' || e.key === ' ') { e.preventDefault(); nextSlide(); }
   892→      if (e.key === 'ArrowUp') { e.preventDefault(); prevSlide(); }
