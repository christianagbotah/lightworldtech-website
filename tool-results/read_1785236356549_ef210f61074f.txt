    62→function useTypewriter(messages: string[], speed = 45) {
    63→  const [text, setText] = useState('');
    64→  const [msgIdx, setMsgIdx] = useState(0);
    65→  const [deleting, setDeleting] = useState(false);
    66→
    67→  useEffect(() => {
    68→    const current = messages[msgIdx];
    69→    const timeout = setTimeout(() => {
    70→      if (!deleting) {
    71→        const next = current.slice(0, text.length + 1);
    72→        setText(next);
    73→        if (next.length === current.length) {
    74→          setTimeout(() => setDeleting(true), 2800);
    75→        }
    76→      } else {
    77→        const next = text.slice(0, -1);
    78→        setText(next);
    79→        if (next.length === 0) {
    80→          setDeleting(false);
    81→          setMsgIdx((prev) => (prev + 1) % messages.length);
    82→        }
    83→      }
    84→    }, deleting ? 22 : speed);
    85→    return () => clearTimeout(timeout);
    86→  }, [text, msgIdx, deleting, messages, speed]);
    87→
    88→  return text;
    89→}

    91→function useCountUp(end: number, duration = 2000, decimals = 0, delay = 600) {
    92→  const [count, setCount] = useState(0);
    93→  const [started, setStarted] = useState(false);
    94→
    95→  useEffect(() => {
    96→    const t = setTimeout(() => setStarted(true), delay);
    97→    return () => clearTimeout(t);
    98→  }, [delay]);
    99→
   100→  useEffect(() => {
   101→    if (!started) return;
   102→    const start = Date.now();
   103→    const interval = setInterval(() => {
   104→      const progress = Math.min((Date.now() - start) / duration, 1);
   105→      const eased = 1 - Math.pow(1 - progress, 3);
   106→      setCount(Number((eased * end).toFixed(decimals)));
   107→      if (progress >= 1) clearInterval(interval);
   108→    }, 16);
   109→    return () => clearInterval(interval);
   110→  }, [started, end, duration, decimals]);
   111→
   112→  return count;
   113→}

   119→/** Deterministic seeded PRNG (mulberry32) — keeps SSR & client output identical. */
   120→function mulberry32(seed: number) {
   121→  return function () {
   122→    let t = (seed += 0x6d2b79f5);
   123→    t = Math.imul(t ^ (t >>> 15), t | 1);
   124→    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
   125→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
   126→  };
   127→}

   780→const slideVariants = {
   781→  enter: { opacity: 0, scale: 0.98 },
   782→  center: { opacity: 1, scale: 1 },
   783→  exit: { opacity: 0, scale: 1.02 },
   784→};
   785→
   786→export default function HomePage() {
   787→  const [currentSlide, setCurrentSlide] = useState(0);
   788→  const [isTransitioning, setIsTransitioning] = useState(false);
   789→  const [isPaused, setIsPaused] = useState(false);
   790→  const touchStartY = useRef(0);
   791→  const containerRef = useRef<HTMLDivElement>(null);
   792→  const lastAutoTime = useRef(Date.now());
   793→
   794→  useSEO({
   795→    title: 'Lightworld Technologies — Powering Digital Innovation',
   796→    description: 'Leading IT solutions provider in Ghana. Web development, cloud services, cybersecurity, data analytics, and enterprise software.',
   797→    keywords: ['IT company Ghana', 'Lightworld Technologies', 'web development Ghana', 'cybersecurity Africa', 'cloud services'],
   798→  });
   799→
   800→  // Auto-advance every 6 seconds
   801→  useEffect(() => {
   802→    if (isPaused) return;
   803→    const interval = setInterval(() => {
   804→      if (!isPaused && !isTransitioning) {
   805→        setCurrentSlide((prev) => (prev + 1) % SLIDES.length);
   806→      }
   807→    }, 7000);
   808→    return () => clearInterval(interval);
   809→  }, [isPaused, isTransitioning]);
   810→
   811→  const goToSlide = useCallback((index: number) => {
   812→    if (isTransitioning || index === currentSlide) return;
   813→    setIsTransitioning(true);
   814→    setCurrentSlide(index);
   815→    setTimeout(() => setIsTransitioning(false), 700);
   816→  }, [currentSlide, isTransitioning]);
   817→
   818→  const nextSlide = useCallback(() => goToSlide((currentSlide + 1) % SLIDES.length), [currentSlide, goToSlide]);
   819→  const prevSlide = useCallback(() => goToSlide((currentSlide - 1 + SLIDES.length) % SLIDES.length), [currentSlide, goToSlide]);
   820→
   821→  // Wheel handler
   822→  useEffect(() => {
   823→    let lastWheel = 0;
   824→    const handleWheel = (e: WheelEvent) => {
   825→      e.preventDefault();
   826→      const now = Date.now();
   827→      if (now - lastWheel < 1000) return; // Throttle
   828→      lastWheel = now;
   829→
   830→      if (Math.abs(e.deltaY) > 30) {
   831→        if (e.deltaY > 0) nextSlide();
   832→        else prevSlide();
   833→      }
   834→    };
   835→
   836→    const el = containerRef.current;
   837→    if (el) {
   838→      el.addEventListener('wheel', handleWheel, { passive: false });
   839→      return () => el.removeEventListener('wheel', handleWheel);
   840→    }
   841→  }, [nextSlide, prevSlide]);
   842→
   843→  // Touch handlers
   844→  const handleTouchStart = useCallback((e: React.TouchEvent) => {
   845→    touchStartY.current = e.touches[0].clientY;
   846→  }, []);
   847→
   848→  const handleTouchEnd = useCallback((e: React.TouchEvent) => {
   849→    const diff = touchStartY.current - e.changedTouches[0].clientY;
   850→    if (Math.abs(diff) > 50) {
   851→      if (diff > 0) nextSlide();
   852→      else prevSlide();
   853→    }
   854→  }, [nextSlide, prevSlide]);
   855→
   856→  // Keyboard
   857→  useEffect(() => {
   858→    const handleKey = (e: KeyboardEvent) => {
   859→      if (e.key === 'ArrowDown' || e.key === ' ') { e.preventDefault(); nextSlide(); }
   860→      if (e.key === 'ArrowUp') { e.preventDefault(); prevSlide(); }
   861→    };
   862→    window.addEventListener('keydown', handleKey);
   863→    return () => window.removeEventListener('keydown', handleKey);
   864→  }, [nextSlide, prevSlide]);
   865→
   866→  const ActiveSlideComponent = SLIDES[currentSlide].component;
   867→
   868→  return (
   869→    <div
   870→      ref={containerRef}
   871→      className="h-screen w-screen overflow-hidden bg-[#050810] relative select-none"
   872→      onMouseEnter={() => setIsPaused(true)}
   873→      onMouseLeave={() => setIsPaused(false)}
   874→      onTouchStart={handleTouchStart}
   875→      onTouchEnd={handleTouchEnd}
   876→    >
   877→      {/* ── Slide Content ── */}
   878→      <AnimatePresence mode="wait">
   879→        <motion.div
   880→          key={currentSlide}
   881→          variants={slideVariants}
   882→          initial="enter"
   883→          animate="center"
   884→          exit="exit"
   885→          transition={{ duration: 0.6, ease: [0.22, 1, 0.36, 1] }}
   886→          className="absolute inset-0"
   887→        >
   888→          <ActiveSlideComponent />
   889→        </motion.div>
   890→      </AnimatePresence>
   891→
   892→      {/* ── Navigation Dots (right side) ── */}
   893→      <div className="fixed right-4 lg:right-8 top-1/2 -translate-y-1/2 z-30 flex flex-col gap-2.5">
   894→        {SLIDES.map((slide, i) => (
   895→          <button
   896→            key={i}
   897→            onClick={() => goToSlide(i)}
