   475→              className="absolute top-3 right-3 size-7 rounded-full bg-black/20 hover:bg-black/40 text-white flex items-center justify-center transition-colors z-10"
   476→              aria-label="Close"
   477→            >
   478→              <X className="size-3.5" />
   479→            </button>
   480→          </div>
   481→
   482→          <div className="p-6">
   483→            <DialogHeader>
   484→              <DialogTitle className="text-lg dark:text-white text-slate-900">{selectedJob?.title}</DialogTitle>
   485→              <DialogDescription className="text-sm dark:text-white/50 text-slate-500">
   486→                {selectedJob?.department} · {selectedJob?.location} · {selectedJob?.type}
   487→              </DialogDescription>
   488→            </DialogHeader>
   489→
   490→            <ApplicationForm
   491→              jobTitle={selectedJob?.title || ''}

   494→          </div>
   495→        </DialogContent>
   496→      </Dialog>
   497→
   498→      {/* CTA */}
   499→      <CTASection />
   500→    </main>
   501→  );

   504→function ApplicationForm({ jobTitle, onSuccess }: { jobTitle: string; onSuccess: () => void }) {
   505→  const [formData, setFormData] = useState({
   506→    name: '',
   507→    email: '',
   508→    phone: '',
   509→    position: jobTitle,
   510→    coverLetter: '',
   511→  });
   512→  const [resumeFile, setResumeFile] = useState<File | null>(null);
   513→  const [submitting, setSubmitting] = useState(false);
   514→  const [submitted, setSubmitted] = useState(false);
   515→
   516→  const handleChange = (e: React.ChangeEvent<HTMLInputElement | HTMLTextAreaElement>) => {
   517→    setFormData(prev => ({ ...prev, [e.target.name]: e.target.value }));
   518→  };
   519→
   520→  const handleFileChange = (e: React.ChangeEvent<HTMLInputElement>) => {
   521→    if (e.target.files && e.target.files[0]) {
   522→      setResumeFile(e.target.files[0]);
   523→    }
   524→  };
   525→
   526→  const handleSubmit = async (e: React.FormEvent) => {
   527→    e.preventDefault();
   528→    if (!formData.name || !formData.email || !formData.position) return;
   529→
   530→    setSubmitting(true);
   531→    try {
   532→      // Upload resume if present
   533→      let resumeUrl = '';
   534→      if (resumeFile) {
   535→        const uploadForm = new FormData();
   536→        uploadForm.append('file', resumeFile);
   537→        const uploadRes = await fetch('/api/upload', {
   538→          method: 'POST',
   539→          body: uploadForm,
   540→        });
   541→        const uploadData = await uploadRes.json();
   542→        if (uploadData.success && uploadData.url) {
   543→          resumeUrl = uploadData.url;
   544→        }
   545→      }
   546→
   547→      const res = await fetch('/api/careers/apply', {
   548→        method: 'POST',
   549→        headers: { 'Content-Type': 'application/json' },
   550→        body: JSON.stringify({
   551→          ...formData,
   552→          resumeUrl,
   553→        }),
   554→      });
   555→
   556→      if (res.ok) {
   557→        toast.success('Application submitted!', {
   558→          description: `We've received your application for ${jobTitle}.`,
   559→        });
   560→        setSubmitted(true);
   561→        setTimeout(onSuccess, 1500);
   562→      } else {
   563→        toast.error('Submission failed', {
   564→          description: 'Please try again or contact us directly.',
   565→        });
   566→      }
   567→    } catch {
   568→      toast.error('Network error', {
   569→        description: 'Please check your connection and try again.',
   570→      });
   571→    } finally {
   572→      setSubmitting(false);
   573→    }
   574→  };
   575→
   576→  if (submitted) {
   577→    return (
   578→      <motion.div
   579→        className="text-center py-8"
   580→        initial={{ opacity: 0, scale: 0.95 }}
   581→        animate={{ opacity: 1, scale: 1 }}
   582→      >
   583→        <motion.div
   584→          className="size-16 rounded-full bg-emerald-500/10 border border-emerald-500/20 flex items-center justify-center mx-auto mb-4"
   585→          initial={{ scale: 0 }}
   586→          animate={{ scale: 1 }}
   587→          transition={{ type: 'spring', stiffness: 200, damping: 15 }}
   588→        >
   589→          <CheckCircle2 className="size-8 text-emerald-400" />
   590→        </motion.div>
   591→        <h3 className="text-lg font-semibold dark:text-white text-slate-900 mb-1">Application Received!</h3>
   592→        <p className="text-sm dark:text-white/40 text-slate-500">
   593→          Thank you for your interest. Our team will review your application and get back to you soon.
   594→        </p>
   595→      </motion.div>
   596→    );
   597→  }
   598→
   599→  return (
   600→    <form onSubmit={handleSubmit} className="space-y-4 mt-4">
   601→      <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   602→        <div className="space-y-1.5">
   603→          <Label htmlFor="app-name" className="text-xs font-medium dark:text-white/60 text-slate-500">Full Name *</Label>
   604→          <Input
   605→            id="app-name"
   606→            name="name"
   607→            value={formData.name}
   608→            onChange={handleChange}
   609→            placeholder="Your full name"
   610→            required
   611→            disabled={submitting}
   612→            className="h-9 dark:bg-white/[0.04] bg-slate-50 dark:border-white/[0.08] border-slate-200 dark:text-white text-slate-900 text-sm dark:placeholder:text-white/25 placeholder:text-slate-400 focus-visible:ring-emerald-500/30 focus-visible:border-emerald-500/40"
   613→          />
   614→        </div>
   615→        <div className="space-y-1.5">
   616→          <Label htmlFor="app-email" className="text-xs font-medium dark:text-white/60 text-slate-500">Email *</Label>
   617→          <Input
   618→            id="app-email"
   619→            name="email"
   620→            type="email"
   621→            value={formData.email}
   622→            onChange={handleChange}
   623→            placeholder="you@example.com"
   624→            required
   625→            disabled={submitting}
   626→            className="h-9 dark:bg-white/[0.04] bg-slate-50 dark:border-white/[0.08] border-slate-200 dark:text-white text-slate-900 text-sm dark:placeholder:text-white/25 placeholder:text-slate-400 focus-visible:ring-emerald-500/30 focus-visible:border-emerald-500/40"
   627→          />
   628→        </div>
   629→      </div>
   630→      <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   631→        <div className="space-y-1.5">
   632→          <Label htmlFor="app-phone" className="text-xs font-medium dark:text-white/60 text-slate-500">Phone</Label>
   633→          <Input
   634→            id="app-phone"
   635→            name="phone"
   636→            type="tel"
   637→            value={formData.phone}
   638→            onChange={handleChange}
   639→            placeholder="+233 XX XXX XXXX"
   640→            disabled={submitting}
   641→            className="h-9 dark:bg-white/[0.04] bg-slate-50 dark:border-white/[0.08] border-slate-200 dark:text-white text-slate-900 text-sm dark:placeholder:text-white/25 placeholder:text-slate-400 focus-visible:ring-emerald-500/30 focus-visible:border-emerald-500/40"
   642→          />
   643→        </div>
   644→        <div className="space-y-1.5">
   645→          <Label htmlFor="app-position" className="text-xs font-medium dark:text-white/60 text-slate-500">Position *</Label>
   646→          <Input
   647→            id="app-position"
   648→            name="position"
   649→            value={formData.position}
   650→            onChange={handleChange}
   651→            required
   652→            disabled={submitting}
   653→            className="h-9 dark:bg-white/[0.04] bg-slate-50 dark:border-white/[0.08] border-slate-200 dark:text-white text-slate-900 text-sm dark:placeholder:text-white/25 placeholder:text-slate-400 focus-visible:ring-emerald-500/30 focus-visible:border-emerald-500/40"
   654→          />
   655→        </div>
   656→      </div>
   657→      <div className="space-y-1.5">
   658→        <Label htmlFor="app-cover" className="text-xs font-medium dark:text-white/60 text-slate-500">Cover Letter</Label>
   659→        <Textarea
   660→          id="app-cover"
   661→          name="coverLetter"
   662→          value={formData.coverLetter}
   663→          onChange={handleChange}
   664→          placeholder="Tell us why you're a great fit for this role..."
   665→          rows={3}
   666→          disabled={submitting}
   667→          className="dark:bg-white/[0.04] bg-slate-50 dark:border-white/[0.08] border-slate-200 dark:text-white text-slate-900 text-sm dark:placeholder:text-white/25 placeholder:text-slate-400 focus-visible:ring-emerald-500/30 focus-visible:border-emerald-500/40 resize-none"
   668→        />
   669→      </div>
   670→      <div className="space-y-1.5">
   671→        <Label htmlFor="app-resume" className="text-xs font-medium dark:text-white/60 text-slate-500">Resume / CV</Label>
   672→        <div className="flex items-center gap-3">
   673→          <label
   674→            htmlFor="app-resume"
   675→            className="flex items-center gap-2 px-3 py-1.5 rounded-lg border border-dashed dark:border-white/[0.08] border-slate-200 hover:border-emerald-500/30 cursor-pointer transition-colors text-xs dark:text-white/35 text-slate-600 hover:text-emerald-400"
   676→          >
   677→            <Upload className="size-3.5" />
   678→            {resumeFile ? resumeFile.name : 'Upload Resume (PDF, DOC)'}
   679→            <input
   680→              id="app-resume"
   681→              type="file"
   682→              accept=".pdf,.doc,.docx"
   683→              onChange={handleFileChange}
   684→              disabled={submitting}
   685→              className="sr-only"
   686→            />
   687→          </label>
   688→          {resumeFile && (
   689→            <button
   690→              type="button"
   691→              onClick={() => setResumeFile(null)}
   692→              className="text-xs dark:text-white/30 text-slate-400 hover:text-red-400 transition-colors"
   693→            >
   694→              Remove
   695→            </button>
   696→          )}
   697→        </div>
   698→      </div>
   699→      <Button
   700→        type="submit"
   701→        disabled={submitting || !formData.name || !formData.email}
   702→        className="w-full bg-gradient-to-r from-emerald-600 to-emerald-500 hover:from-emerald-500 hover:to-emerald-400 text-white text-sm font-semibold shadow-md hover:shadow-lg hover:shadow-emerald-500/20 transition-all"
   703→      >
   704→        {submitting ? (
   705→          <>
   706→            <Loader2 className="size-4 mr-2 animate-spin" />
   707→            Submitting...
   708→          </>
   709→        ) : (
   710→          <>
   711→            Submit Application
   712→            <Send className="size-4 ml-2" />
   713→          </>
   714→        )}
   715→      </Button>
   716→    </form>
   717→  );
   718→}
