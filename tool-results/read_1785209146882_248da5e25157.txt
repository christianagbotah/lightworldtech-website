   570→function ApplicationForm({ jobTitle, onSuccess }: { jobTitle: string; onSuccess: () => void }) {
   571→  const [formData, setFormData] = useState({
   572→    name: '',
   573→    email: '',
   574→    phone: '',
   575→    position: jobTitle,
   576→    coverLetter: '',
   577→  });
   578→  const [resumeFile, setResumeFile] = useState<File | null>(null);
   579→  const [submitting, setSubmitting] = useState(false);
   580→  const [submitted, setSubmitted] = useState(false);
   581→
   582→  const handleChange = (e: React.ChangeEvent<HTMLInputElement | HTMLTextAreaElement>) => {
   583→    setFormData(prev => ({ ...prev, [e.target.name]: e.target.value }));
   584→  };
   585→
   586→  const handleFileChange = (e: React.ChangeEvent<HTMLInputElement>) => {
   587→    if (e.target.files && e.target.files[0]) {
   588→      setResumeFile(e.target.files[0]);
   589→    }
   590→  };
   591→
   592→  const handleSubmit = async (e: React.FormEvent) => {
   593→    e.preventDefault();
   594→    if (!formData.name || !formData.email || !formData.position) return;
   595→
   596→    setSubmitting(true);
   597→    try {
   598→      // Upload resume if present
   599→      let resumeUrl = '';
   600→      if (resumeFile) {
   601→        const uploadForm = new FormData();
   602→        uploadForm.append('file', resumeFile);
   603→        const uploadRes = await fetch('/api/upload', {
   604→          method: 'POST',
   605→          body: uploadForm,
   606→        });
   607→        const uploadData = await uploadRes.json();
   608→        if (uploadData.success && uploadData.url) {
   609→          resumeUrl = uploadData.url;
   610→        }
   611→      }
   612→
   613→      const res = await fetch('/api/careers/apply', {
   614→        method: 'POST',
   615→        headers: { 'Content-Type': 'application/json' },
   616→        body: JSON.stringify({
   617→          ...formData,
   618→          resumeUrl,
   619→        }),
   620→      });
   621→
   622→      if (res.ok) {
   623→        toast.success('Application submitted!', {
   624→          description: `We've received your application for ${jobTitle}.`,
   625→        });
   626→        setSubmitted(true);
   627→        setTimeout(onSuccess, 1500);
   628→      } else {
   629→        toast.error('Submission failed', {
   630→          description: 'Please try again or contact us directly.',
   631→        });
   632→      }
   633→    } catch {
   634→      toast.error('Network error', {
   635→        description: 'Please check your connection and try again.',
   636→      });
   637→    } finally {
   638→      setSubmitting(false);
   639→    }
   640→  };
   641→
   642→  if (submitted) {
   643→    return (
   644→      <motion.div
   645→        className="text-center py-8"
   646→        initial={{ opacity: 0, scale: 0.95 }}
   647→        animate={{ opacity: 1, scale: 1 }}
   648→      >
   649→        <motion.div
   650→          className="size-16 rounded-full bg-amber-100 dark:bg-amber-900/30 flex items-center justify-center mx-auto mb-4"
   651→          initial={{ scale: 0 }}
   652→          animate={{ scale: 1 }}
   653→          transition={{ type: 'spring', stiffness: 200, damping: 15 }}
   654→        >
   655→          <CheckCircle2 className="size-8 text-emerald-600 dark:text-amber-400" />
   656→        </motion.div>
   657→        <h3 className="text-lg font-semibold text-slate-900 dark:text-white mb-1">Application Received!</h3>
   658→        <p className="text-sm text-slate-500 dark:text-slate-400">
   659→          Thank you for your interest. Our team will review your application and get back to you soon.
   660→        </p>
   661→      </motion.div>
   662→    );
   663→  }
   664→
   665→  return (
   666→    <form onSubmit={handleSubmit} className="space-y-4 mt-4">
   667→      <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   668→        <div className="space-y-1.5">
   669→          <Label htmlFor="app-name" className="text-sm font-medium">Full Name *</Label>
   670→          <Input
   671→            id="app-name"
   672→            name="name"
   673→            value={formData.name}
   674→            onChange={handleChange}
   675→            placeholder="Your full name"
   676→            required
   677→            disabled={submitting}
   678→            className="h-10 focus-visible:ring-emerald-500/30 focus-visible:border-amber-400 dark:focus-visible:border-emerald-500"
   679→          />
   680→        </div>
   681→        <div className="space-y-1.5">
   682→          <Label htmlFor="app-email" className="text-sm font-medium">Email *</Label>
   683→          <Input
   684→            id="app-email"
   685→            name="email"
   686→            type="email"
   687→            value={formData.email}
   688→            onChange={handleChange}
   689→            placeholder="you@example.com"
   690→            required
   691→            disabled={submitting}
   692→            className="h-10 focus-visible:ring-emerald-500/30 focus-visible:border-amber-400 dark:focus-visible:border-emerald-500"
   693→          />
   694→        </div>
   695→      </div>
   696→      <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   697→        <div className="space-y-1.5">
   698→          <Label htmlFor="app-phone" className="text-sm font-medium">Phone</Label>
   699→          <Input
   700→            id="app-phone"
   701→            name="phone"
   702→            type="tel"
   703→            value={formData.phone}
   704→            onChange={handleChange}
   705→            placeholder="+233 XX XXX XXXX"
   706→            disabled={submitting}
   707→            className="h-10 focus-visible:ring-emerald-500/30 focus-visible:border-amber-400 dark:focus-visible:border-emerald-500"
   708→          />
   709→        </div>
   710→        <div className="space-y-1.5">
   711→          <Label htmlFor="app-position" className="text-sm font-medium">Position *</Label>
   712→          <Input
   713→            id="app-position"
   714→            name="position"
   715→            value={formData.position}
   716→            onChange={handleChange}
   717→            required
   718→            disabled={submitting}
   719→            className="h-10 focus-visible:ring-emerald-500/30 focus-visible:border-amber-400 dark:focus-visible:border-emerald-500"
   720→          />
   721→        </div>
   722→      </div>
   723→      <div className="space-y-1.5">
   724→        <Label htmlFor="app-cover" className="text-sm font-medium">Cover Letter</Label>
   725→        <Textarea
   726→          id="app-cover"
   727→          name="coverLetter"
   728→          value={formData.coverLetter}
   729→          onChange={handleChange}
   730→          placeholder="Tell us why you're a great fit for this role..."
   731→          rows={4}
   732→          disabled={submitting}
   733→          className="focus-visible:ring-emerald-500/30 focus-visible:border-amber-400 dark:focus-visible:border-emerald-500 resize-none"
   734→        />
   735→      </div>
   736→      <div className="space-y-1.5">
   737→        <Label htmlFor="app-resume" className="text-sm font-medium">Resume / CV</Label>
   738→        <div className="flex items-center gap-3">
   739→          <label
   740→            htmlFor="app-resume"
   741→            className="flex items-center gap-2 px-4 py-2 rounded-lg border border-dashed border-slate-300 dark:border-slate-600 hover:border-amber-400 dark:hover:border-emerald-500 cursor-pointer transition-colors text-sm text-slate-500 dark:text-slate-400 hover:text-emerald-600 dark:hover:text-amber-400"
   742→          >
   743→            <Upload className="size-4" />
   744→            {resumeFile ? resumeFile.name : 'Upload Resume (PDF, DOC)'}
   745→            <input
   746→              id="app-resume"
   747→              type="file"
   748→              accept=".pdf,.doc,.docx"
   749→              onChange={handleFileChange}
   750→              disabled={submitting}
   751→              className="sr-only"
   752→            />
   753→          </label>
   754→          {resumeFile && (
   755→            <button
   756→              type="button"
   757→              onClick={() => setResumeFile(null)}
   758→              className="text-xs text-slate-400 hover:text-red-500 transition-colors"
   759→            >
   760→              Remove
   761→            </button>
   762→          )}
   763→        </div>
   764→      </div>
   765→      <Button
   766→        type="submit"
   767→        disabled={submitting || !formData.name || !formData.email}
   768→        className="w-full bg-emerald-600 hover:bg-emerald-700 text-white shadow-md hover:shadow-lg transition-all"
   769→      >
   770→        {submitting ? (
   771→          <>
   772→            <Loader2 className="size-4 mr-2 animate-spin" />
   773→            Submitting...
   774→          </>
   775→        ) : (
   776→          <>
   777→            Submit Application
   778→            <Send className="size-4 ml-2" />
   779→          </>
   780→        )}
   781→      </Button>
   782→    </form>
   783→  );
   784→}
