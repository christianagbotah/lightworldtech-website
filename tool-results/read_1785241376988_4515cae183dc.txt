    62→    62→function useTypewriter(messages: string[], speed = 45) {
    63→    63→  const [text, setText] = useState('');
    64→    64→  const [msgIdx, setMsgIdx] = useState(0);
    65→    65→  const [deleting, setDeleting] = useState(false);
    66→    66→
    67→    67→  useEffect(() => {
    68→    68→    const current = messages[msgIdx];
    69→    69→    const timeout = setTimeout(() => {
    70→    70→      if (!deleting) {
    71→    71→        const next = current.slice(0, text.length + 1);
    72→    72→        setText(next);
    73→    73→        if (next.length === current.length) {
    74→    74→          setTimeout(() => setDeleting(true), 2800);
    75→    75→        }
    76→    76→      } else {
    77→    77→        const next = text.slice(0, -1);
    78→    78→        setText(next);
    79→    79→        if (next.length === 0) {
    80→    80→          setDeleting(false);
    81→    81→          setMsgIdx((prev) => (prev + 1) % messages.length);
    82→    82→        }
    83→    83→      }
    84→    84→    }, deleting ? 22 : speed);
    85→    85→    return () => clearTimeout(timeout);
    86→    86→  }, [text, msgIdx, deleting, messages, speed]);
    87→    87→
    88→    88→  return text;
    89→    89→}

    91→    91→function useCountUp(end: number, duration = 2000, decimals = 0, delay = 600) {
    92→    92→  const [count, setCount] = useState(0);
    93→    93→  const [started, setStarted] = useState(false);
    94→    94→
    95→    95→  useEffect(() => {
    96→    96→    const t = setTimeout(() => setStarted(true), delay);
    97→    97→    return () => clearTimeout(t);
    98→    98→  }, [delay]);
    99→    99→
   100→   100→  useEffect(() => {
   101→   101→    if (!started) return;
   102→   102→    const start = Date.now();
   103→   103→    const interval = setInterval(() => {
   104→   104→      const progress = Math.min((Date.now() - start) / duration, 1);
   105→   105→      const eased = 1 - Math.pow(1 - progress, 3);
   106→   106→      setCount(Number((eased * end).toFixed(decimals)));
   107→   107→      if (progress >= 1) clearInterval(interval);
   108→   108→    }, 16);
   109→   109→    return () => clearInterval(interval);
   110→   110→  }, [started, end, duration, decimals]);
   111→   111→
   112→   112→  return count;
   113→   113→}

   119→   119→/** Deterministic seeded PRNG (mulberry32) — keeps SSR & client output identical. */
   120→   120→function mulberry32(seed: number) {
   121→   121→  return function () {
   122→   122→    let t = (seed += 0x6d2b79f5);
   123→   123→    t = Math.imul(t ^ (t >>> 15), t | 1);
   124→   124→    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
   125→   125→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
   126→   126→  };
   127→   127→}
