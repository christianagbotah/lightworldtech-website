   396→                    {selectedProject.featured && <Badge className="bg-amber-500/10 text-amber-300">Featured</Badge>}
   397→                  </div>
   398→                  <Button size="sm" onClick={() => { const idx = filtered.findIndex(p => p.id === selectedProject.id); setLightboxIndex(idx >= 0 ? idx : 0); setLightboxOpen(true); }} className="bg-emerald-500 hover:bg-emerald-400 text-white text-xs">
   399→                    <ExternalLink className="size-3 mr-1" /> Full View
   400→                  </Button>
   401→                </div>
   402→                <DialogHeader className="mt-1"><DialogTitle className="text-xl dark:text-white text-slate-900">{selectedProject.title}</DialogTitle></DialogHeader>
   403→                <DialogDescription className="text-sm dark:text-white/60 text-slate-500 leading-relaxed">{selectedProject.fullDescription || selectedProject.description}</DialogDescription>
   404→
   405→                <div>
   406→                  <h4 className="text-sm font-semibold dark:text-white text-slate-900 mb-2">Technologies</h4>
   407→                  <div className="flex flex-wrap gap-1.5">
   408→                    {selectedProject.tags?.map((tag: string) => (
   409→                      <Badge key={tag} variant="secondary" className="text-xs dark:bg-white/[0.06] bg-slate-100 dark:text-white/40 text-slate-500">{tag}</Badge>
   410→                    ))}
   411→                  </div>
   412→                </div>
   413→
   414→                <div className="flex gap-3 pt-2">
   415→                  <Button className="bg-emerald-500 hover:bg-emerald-400 text-white flex-1"><ExternalLink className="size-4 mr-2" /> Visit Project</Button>
   416→                  <Button variant="outline" onClick={() => setSelectedProject(null)} className="flex-1">Close</Button>
   417→                </div>
   418→              </div>
   419→            </>
   420→          )}
   421→        </DialogContent>
   422→      </Dialog>
   423→
   424→      {/* Image Lightbox */}
   425→      <ImageLightbox
   426→        items={filtered.map(p => ({
   427→          id: p.id,
   428→          title: p.title,
   429→          description: p.fullDescription || p.description,
   430→          category: p.category,
   431→          tags: p.tags,
   432→          featured: p.featured,
   433→        }))}
   434→        currentIndex={lightboxIndex}
   435→        isOpen={lightboxOpen}
   436→        onClose={() => setLightboxOpen(false)}
   437→        onNavigate={(index) => {
   438→          setLightboxIndex(index);
   439→          setSelectedProject(filtered[index] || null);
   440→        }}
   441→      />
   442→
   443→      {/* CTA */}
   444→      <CTASection />
   445→    </main>
   446→  );
