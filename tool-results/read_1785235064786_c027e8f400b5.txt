    62→function useTypewriter(messages: string[], speed = 45) {
    63→  const [text, setText] = useState('');
    64→  const [msgIdx, setMsgIdx] = useState(0);
    65→  const [deleting, setDeleting] = useState(false);
    66→
    67→  useEffect(() => {
    68→    const current = messages[msgIdx];
    69→    const timeout = setTimeout(() => {
    70→      if (!deleting) {
    71→        const next = current.slice(0, text.length + 1);
    72→        setText(next);
    73→        if (next.length === current.length) {
    74→          setTimeout(() => setDeleting(true), 2800);
    75→        }
    76→      } else {
    77→        const next = text.slice(0, -1);
    78→        setText(next);
    79→        if (next.length === 0) {
    80→          setDeleting(false);
    81→          setMsgIdx((prev) => (prev + 1) % messages.length);
    82→        }
    83→      }
    84→    }, deleting ? 22 : speed);
    85→    return () => clearTimeout(timeout);
    86→  }, [text, msgIdx, deleting, messages, speed]);
    87→
    88→  return text;
    89→}

    91→function useCountUp(end: number, duration = 2000, decimals = 0, delay = 600) {
    92→  const [count, setCount] = useState(0);
    93→  const [started, setStarted] = useState(false);
    94→
    95→  useEffect(() => {
    96→    const t = setTimeout(() => setStarted(true), delay);
    97→    return () => clearTimeout(t);
    98→  }, [delay]);
    99→
   100→  useEffect(() => {
   101→    if (!started) return;
   102→    const start = Date.now();
   103→    const interval = setInterval(() => {
   104→      const progress = Math.min((Date.now() - start) / duration, 1);
   105→      const eased = 1 - Math.pow(1 - progress, 3);
   106→      setCount(Number((eased * end).toFixed(decimals)));
   107→      if (progress >= 1) clearInterval(interval);
   108→    }, 16);
   109→    return () => clearInterval(interval);
   110→  }, [started, end, duration, decimals]);
   111→
   112→  return count;
   113→}

   773→  center: { opacity: 1, scale: 1 },
   774→  exit: { opacity: 0, scale: 1.02 },
   775→};
   776→
   777→export default function HomePage() {
   778→  const [currentSlide, setCurrentSlide] = useState(0);
   779→  const [isTransitioning, setIsTransitioning] = useState(false);
   780→  const [isPaused, setIsPaused] = useState(false);
   781→  const touchStartY = useRef(0);
   782→  const containerRef = useRef<HTMLDivElement>(null);
   783→  const lastAutoTime = useRef(Date.now());
   784→
   785→  useSEO({
   786→    title: 'Lightworld Technologies — Powering Digital Innovation',
   787→    description: 'Leading IT solutions provider in Ghana. Web development, cloud services, cybersecurity, data analytics, and enterprise software.',
   788→    keywords: ['IT company Ghana', 'Lightworld Technologies', 'web development Ghana', 'cybersecurity Africa', 'cloud services'],
   789→  });
   790→
   791→  // Auto-advance every 6 seconds
   792→  useEffect(() => {
   793→    if (isPaused) return;
   794→    const interval = setInterval(() => {
   795→      if (!isPaused && !isTransitioning) {
   796→        setCurrentSlide((prev) => (prev + 1) % SLIDES.length);
   797→      }
   798→    }, 7000);
   799→    return () => clearInterval(interval);
   800→  }, [isPaused, isTransitioning]);
   801→
   802→  const goToSlide = useCallback((index: number) => {
   803→    if (isTransitioning || index === currentSlide) return;
   804→    setIsTransitioning(true);
   805→    setCurrentSlide(index);
   806→    setTimeout(() => setIsTransitioning(false), 700);
   807→  }, [currentSlide, isTransitioning]);
   808→
   809→  const nextSlide = useCallback(() => goToSlide((currentSlide + 1) % SLIDES.length), [currentSlide, goToSlide]);
   810→  const prevSlide = useCallback(() => goToSlide((currentSlide - 1 + SLIDES.length) % SLIDES.length), [currentSlide, goToSlide]);
   811→
   812→  // Wheel handler
   813→  useEffect(() => {
   814→    let lastWheel = 0;
   815→    const handleWheel = (e: WheelEvent) => {
   816→      e.preventDefault();
   817→      const now = Date.now();
   818→      if (now - lastWheel < 1000) return; // Throttle
   819→      lastWheel = now;
   820→
   821→      if (Math.abs(e.deltaY) > 30) {
   822→        if (e.deltaY > 0) nextSlide();
   823→        else prevSlide();
   824→      }
   825→    };
   826→
   827→    const el = containerRef.current;
   828→    if (el) {
   829→      el.addEventListener('wheel', handleWheel, { passive: false });
   830→      return () => el.removeEventListener('wheel', handleWheel);
   831→    }
   832→  }, [nextSlide, prevSlide]);
   833→
   834→  // Touch handlers
   835→  const handleTouchStart = useCallback((e: React.TouchEvent) => {
   836→    touchStartY.current = e.touches[0].clientY;
   837→  }, []);
   838→
   839→  const handleTouchEnd = useCallback((e: React.TouchEvent) => {
   840→    const diff = touchStartY.current - e.changedTouches[0].clientY;
   841→    if (Math.abs(diff) > 50) {
   842→      if (diff > 0) nextSlide();
   843→      else prevSlide();
   844→    }
   845→  }, [nextSlide, prevSlide]);
   846→
   847→  // Keyboard
   848→  useEffect(() => {
   849→    const handleKey = (e: KeyboardEvent) => {
   850→      if (e.key === 'ArrowDown' || e.key === ' ') { e.preventDefault(); nextSlide(); }
   851→      if (e.key === 'ArrowUp') { e.preventDefault(); prevSlide(); }
   852→    };
   853→    window.addEventListener('keydown', handleKey);
   854→    return () => window.removeEventListener('keydown', handleKey);
   855→  }, [nextSlide, prevSlide]);
   856→
   857→  const ActiveSlideComponent = SLIDES[currentSlide].component;
   858→
   859→  return (
   860→    <div
   861→      ref={containerRef}
   862→      className="h-screen w-screen overflow-hidden bg-[#050810] relative select-none"
   863→      onMouseEnter={() => setIsPaused(true)}
   864→      onMouseLeave={() => setIsPaused(false)}
   865→      onTouchStart={handleTouchStart}
   866→      onTouchEnd={handleTouchEnd}
   867→    >
   868→      {/* ── Slide Content ── */}
   869→      <AnimatePresence mode="wait">
   870→        <motion.div
   871→          key={currentSlide}
   872→          variants={slideVariants}
   873→          initial="enter"
   874→          animate="center"
   875→          exit="exit"
   876→          transition={{ duration: 0.6, ease: [0.22, 1, 0.36, 1] }}
   877→          className="absolute inset-0"
   878→        >
   879→          <ActiveSlideComponent />
   880→        </motion.div>
   881→      </AnimatePresence>
   882→
   883→      {/* ── Navigation Dots (right side) ── */}
   884→      <div className="fixed right-4 lg:right-8 top-1/2 -translate-y-1/2 z-30 flex flex-col gap-2.5">
   885→        {SLIDES.map((slide, i) => (
   886→          <button
   887→            key={i}
   888→            onClick={() => goToSlide(i)}
   889→            className="group flex items-center gap-2.5"
   890→            aria-label={`Go to ${slide.label} slide`}
   891→          >
   892→            {/* Desktop: dot with label */}
   893→            <span className="hidden lg:block text-[9px] text-white/0 group-hover:text-white/40 font-medium transition-all duration-300 text-right w-24 -mr-2.5">
   894→              {slide.label}
   895→            </span>
   896→            <div
   897→              className={cn(
   898→                'transition-all duration-400 rounded-full',
