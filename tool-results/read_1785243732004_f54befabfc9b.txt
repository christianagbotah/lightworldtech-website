    77→  const [hoverRating, setHoverRating] = useState(0);
    78→  const [errors, setErrors] = useState<Partial<Record<keyof ReviewFormData, string>>>({});
    79→  const [submitting, setSubmitting] = useState(false);
    80→  const [submitted, setSubmitted] = useState(false);
    81→
    82→  const validateField = (field: keyof ReviewFormData, value: unknown) => {
    83→    try {
    84→      const partial = reviewSchema.shape[field];
    85→      partial.parse(value);
    86→      setErrors(prev => ({ ...prev, [field]: undefined }));
    87→      return true;
    88→    } catch {
    89→      setErrors(prev => ({ ...prev, [field]: 'Invalid value' }));
    90→      return false;
    91→    }
    92→  };
    93→
    94→  const handleSubmit = async () => {
    95→    try {
    96→      const result = reviewSchema.safeParse(formData);
    97→      if (!result.success) {
    98→        const fieldErrors: Partial<Record<keyof ReviewFormData, string>> = {};
    99→        result.error.issues.forEach((issue) => {
   100→          const field = issue.path[0] as keyof ReviewFormData;
   101→          if (!fieldErrors[field]) {
   102→            fieldErrors[field] = issue.message;
   103→          }
   104→        });
   105→        setErrors(fieldErrors);
   106→        return;
   107→      }
   108→
   109→      setSubmitting(true);
   110→      const response = await fetch('/api/testimonials', {
   111→        method: 'POST',
   112→        headers: { 'Content-Type': 'application/json' },
   113→        body: JSON.stringify(result.data),
   114→      });
   115→
   116→      const data = await response.json();
   117→
   118→      if (data.success) {
   119→        setSubmitted(true);
   120→        toast.success('Thank you for your review! Your feedback is invaluable to us.');
   121→        setTimeout(() => {
   122→          handleClose();
   123→        }, 3000);
   124→      } else {
   125→        toast.error('Failed to submit review. Please try again.');
   126→      }
   127→    } catch {
   128→      toast.error('Something went wrong. Please try again.');
   129→    } finally {
   130→      setSubmitting(false);
   131→    }
   132→  };

   135→    onOpenChange(false);
   136→    setTimeout(() => {
   137→      setFormData({ name: '', company: '', role: '', rating: 0, content: '' });
   138→      setErrors({});
   139→      setSubmitted(false);
   140→      setHoverRating(0);

   165→              <p className="text-slate-500 dark:text-slate-400 text-sm">

   171→              key="form"

   174→              exit={{ opacity: 0, x: -20 }}
   175→            >
   176→              {/* Header */}
   177→              <div className="bg-gradient-to-r from-amber-600 to-amber-500 dark:from-amber-500 dark:to-amber-800 p-6">
   178→                <DialogHeader>

   184→                    Tell us about your experience working with Lightworld Technologies.

   288→                  />
   289→                  <div className="flex justify-between items-center">

   314→            </motion.div>

   331→

   334→      .then((data) => {
   335→        if (Array.isArray(data) && data.length > 0) setTestimonials(data);
   336→      })
   337→      .catch(() => {})
   338→      .finally(() => setLoading(false));
   339→  }, []);
   340→
   341→  // Sync active dot with carousel scroll position
   342→  const syncActiveDot = useCallback(() => {
   343→    const api = carouselApiRef.current;
   344→    if (!api) return;
   345→    const scrollProgress = api.scrollProgress();
   346→    const newIndex = Math.round(scrollProgress * testimonials.length) % testimonials.length;
   347→    setActiveDot(newIndex < 0 ? 0 : newIndex);
   348→  }, [testimonials.length]);
   349→
   350→  // Hook into the carousel API
   351→  const handleApi = useCallback((api: CarouselApi) => {
   352→    if (!api) return;
   353→    carouselApiRef.current = api;
   354→    api.on('select', syncActiveDot);
   355→    api.on('scroll', syncActiveDot);
   356→  }, [syncActiveDot]);
