    30→function formatFileSize(bytes: number): string {
    31→  if (bytes < 1024) return `${bytes} B`;
    32→  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`;
    33→  return `${(bytes / (1024 * 1024)).toFixed(1)} MB`;
    34→}

    36→function getFileIcon(type: string): React.ElementType {
    37→  if (type.startsWith('image/')) return ImageIcon;
    38→  if (type === 'application/pdf') return FileText;
    39→  return File;
    40→}

    42→function getFileColor(type: string): string {
    43→  if (type.startsWith('image/')) return 'bg-amber-100 dark:bg-amber-900/30 text-emerald-600 dark:text-amber-400';
    44→  if (type === 'application/pdf') return 'bg-red-100 dark:bg-red-900/30 text-red-600 dark:text-red-400';
    45→  return 'bg-slate-100 dark:bg-slate-700 text-slate-600 dark:text-slate-300';
    46→}

    54→function CopyButton({ text, label }: { text: string; label: string }) {
    55→  const [copied, setCopied] = useState(false);

    65→  };
    66→
    67→  return (
    68→    <button onClick={handleCopy} className="ml-auto p-1.5 rounded-lg text-slate-400 hover:text-emerald-600 dark:hover:text-amber-400 hover:bg-emerald-50 dark:hover:bg-amber-900/30 transition-all duration-200 shrink-0" aria-label={`Copy ${label}`}>

    74→function FileUploadZone({ files, setFiles }: { files: FileAttachment[]; setFiles: (files: FileAttachment[]) => void }) {
    75→  const inputRef = useRef<HTMLInputElement>(null);
    76→  const [isDragOver, setIsDragOver] = useState(false);
    77→
    78→  const totalSize = useMemo(() => files.reduce((sum, f) => sum + f.file.size, 0), [files]);
    79→
    80→  const addFiles = useCallback((newFiles: FileList | File[]) => {
    81→    const fileArray = Array.from(newFiles);
    82→
    83→    if (files.length + fileArray.length > MAX_FILES) {
    84→      toast.error(`Maximum ${MAX_FILES} files allowed`, { description: `You can upload up to ${MAX_FILES} files at once.` });
    85→      return;
    86→    }
    87→
    88→    const oversized = fileArray.filter(f => f.size > MAX_FILE_SIZE);
    89→    if (oversized.length > 0) {
    90→      toast.error('File too large', { description: `${oversized.map(f => f.name).join(', ')} exceed${oversized.length > 1 ? '' : 's'} the 10MB limit.` });
    91→    }
    92→
    93→    const validFiles = fileArray.filter(f => {
    94→      if (f.size > MAX_FILE_SIZE) return false;
    95→      if (totalSize + f.size > MAX_TOTAL_SIZE) return false;
    96→      if (!ACCEPTED_TYPES.includes(f.type) && !f.name.match(/\.(jpg|jpeg|png|gif|webp|svg|pdf|doc|docx|txt|zip|xls|xlsx)$/i)) return false;
    97→      return true;
    98→    });
    99→
   100→    if (validFiles.length === 0 && fileArray.length > 0) {
   101→      toast.error('Invalid files', { description: 'Please upload images, PDFs, or documents only.' });
   102→      return;
   103→    }
   104→
   105→    const newAttachments: FileAttachment[] = validFiles.map(file => {
   106→      const id = `${Date.now()}-${Math.random().toString(36).slice(2, 9)}`;
   107→      let preview: string | undefined;
   108→
   109→      if (file.type.startsWith('image/')) {
   110→        preview = URL.createObjectURL(file);
   111→      }
   112→
   113→      return { id, file, preview };
   114→    });
   115→
   116→    setFiles([...files, ...newAttachments]);
   117→  }, [files, setFiles, totalSize]);
   118→
   119→  const removeFile = useCallback((id: string) => {
   120→    const file = files.find(f => f.id === id);
   121→    if (file?.preview) URL.revokeObjectURL(file.preview);
   122→    setFiles(files.filter(f => f.id !== id));
   123→  }, [files, setFiles]);
   124→
   125→  const handleDragOver = useCallback((e: React.DragEvent) => {
   126→    e.preventDefault();
   127→    e.stopPropagation();
   128→    setIsDragOver(true);
   129→  }, []);
   130→
   131→  const handleDragLeave = useCallback((e: React.DragEvent) => {
   132→    e.preventDefault();
   133→    e.stopPropagation();
   134→    setIsDragOver(false);
   135→  }, []);
   136→
   137→  const handleDrop = useCallback((e: React.DragEvent) => {
   138→    e.preventDefault();
   139→    e.stopPropagation();
   140→    setIsDragOver(false);
   141→    if (e.dataTransfer.files && e.dataTransfer.files.length > 0) {
   142→      addFiles(e.dataTransfer.files);
   143→    }
   144→  }, [addFiles]);
   145→
   146→  return (
   147→    <div className="space-y-3">
   148→      <Label className="text-slate-900 dark:text-slate-100 font-medium text-sm flex items-center gap-1.5">
   149→        <span className="w-1 h-1 rounded-full bg-amber-500" />
   150→        Attachments
   151→        <span className="text-xs text-slate-400 font-normal">(optional - max {MAX_FILES} files, 10MB each)</span>
   152→      </Label>
   153→
   154→      {/* Drop zone */}
   155→      <div
   156→        className={`relative border-2 border-dashed rounded-xl p-6 text-center transition-all duration-300 cursor-pointer ${
   157→          isDragOver
   158→            ? 'border-emerald-500 bg-amber-50 dark:bg-amber-900/20 scale-[1.01]'
   159→            : 'border-slate-200 dark:border-slate-700 hover:border-amber-400 dark:hover:border-emerald-500 hover:bg-slate-50 dark:hover:bg-slate-800/50'
   160→        }`}
   161→        onDragOver={handleDragOver}
   162→        onDragLeave={handleDragLeave}
   163→        onDrop={handleDrop}
   164→        onClick={() => inputRef.current?.click()}
   165→        role="button"
   166→        tabIndex={0}
   167→        onKeyDown={(e) => { if (e.key === 'Enter' || e.key === ' ') inputRef.current?.click(); }}
   168→        aria-label="Upload files - click or drag and drop"
   169→      >
   170→        <input
   171→          ref={inputRef}
   172→          type="file"
   173→          multiple
   174→          accept=".jpg,.jpeg,.png,.gif,.webp,.svg,.pdf,.doc,.docx,.txt,.zip,.xls,.xlsx"
   175→          onChange={(e) => { if (e.target.files) addFiles(e.target.files); e.target.value = ''; }}
   176→          className="hidden"
   177→        />
   178→        <motion.div animate={isDragOver ? { scale: 1.05 } : { scale: 1 }} transition={{ duration: 0.2 }}>
   179→          <div className={`size-12 rounded-full mx-auto mb-3 flex items-center justify-center transition-colors duration-300 ${
   180→            isDragOver
   181→              ? 'bg-amber-100 dark:bg-amber-800'
   182→              : 'bg-slate-100 dark:bg-slate-800'
   183→          }`}>
   184→            <Upload className={`size-5 transition-colors duration-300 ${isDragOver ? 'text-emerald-600 dark:text-amber-400' : 'text-slate-400'}`} />
   185→          </div>
   186→          <p className={`text-sm font-medium transition-colors duration-300 ${isDragOver ? 'text-emerald-600 dark:text-amber-400' : 'text-slate-600 dark:text-slate-300'}`}>
   187→            {isDragOver ? 'Drop files here' : 'Drag & drop files here'}
   188→          </p>
   189→          <p className="text-xs text-slate-400 mt-1">or click to browse</p>
   190→        </motion.div>
   191→      </div>
   192→
   193→      {/* File list */}
   194→      {files.length > 0 && (
   195→        <div className="space-y-2">
   196→          {files.map((attachment) => {
   197→            const FileIcon = getFileIcon(attachment.file.type);
   198→            const colorClass = getFileColor(attachment.file.type);
   199→            const isOversized = attachment.file.size > MAX_FILE_SIZE;
   200→
   201→            return (
   202→              <motion.div
   203→                key={attachment.id}
   204→                initial={{ opacity: 0, y: -10, height: 0 }}
   205→                animate={{ opacity: 1, y: 0, height: 'auto' }}
   206→                exit={{ opacity: 0, y: -10, height: 0 }}
   207→                className={`flex items-center gap-3 p-3 rounded-lg border transition-all duration-200 ${
   208→                  isOversized ? 'border-red-200 dark:border-red-800 bg-red-50 dark:bg-red-900/10' : 'border-slate-200 dark:border-slate-700 bg-white dark:bg-slate-800'
   209→                }`}
   210→              >
   211→                {/* Preview or icon */}
   212→                {attachment.preview ? (
   213→                  <div className="size-10 rounded-lg overflow-hidden shrink-0 bg-slate-100 dark:bg-slate-700">
   214→                    <img src={attachment.preview} alt={attachment.file.name} className="size-full object-cover" />
   215→                  </div>
   216→                ) : (
   217→                  <div className={`size-10 rounded-lg flex items-center justify-center shrink-0 ${colorClass}`}>
   218→                    <FileIcon className="size-4" />
   219→                  </div>
   220→                )}
   221→
   222→                <div className="flex-1 min-w-0">
   223→                  <p className={`text-xs font-medium truncate ${isOversized ? 'text-red-600 dark:text-red-400' : 'text-slate-700 dark:text-slate-200'}`}>
   224→                    {attachment.file.name}
   225→                  </p>
   226→                  <div className="flex items-center gap-1">
   227→                    <p className={`text-[10px] ${isOversized ? 'text-red-500' : 'text-slate-400'}`}>
   228→                      {formatFileSize(attachment.file.size)}
   229→                    </p>
   230→                    {isOversized && (
   231→                      <span className="text-[10px] text-red-500 flex items-center gap-0.5">
   232→                        <AlertCircle className="size-2.5" />
   233→                        Too large
   234→                      </span>
   235→                    )}
   236→                  </div>
   237→                </div>
   238→
   239→                <button
   240→                  onClick={(e) => { e.stopPropagation(); removeFile(attachment.id); }}
   241→                  className="size-7 rounded-full hover:bg-red-50 dark:hover:bg-red-900/20 text-slate-400 hover:text-red-500 flex items-center justify-center transition-colors shrink-0"
   242→                  aria-label={`Remove ${attachment.file.name}`}
   243→                >
   244→                  <X className="size-3.5" />
   245→                </button>
   246→              </motion.div>
   247→            );
   248→          })}
   249→
   250→          {/* Total size indicator */}
   251→          <div className="flex items-center justify-between px-1">
   252→            <span className="text-[10px] text-slate-400">{files.length} file{files.length !== 1 ? 's' : ''} attached</span>
   253→            <div className="flex items-center gap-2">
   254→              <span className={`text-[10px] ${totalSize > MAX_TOTAL_SIZE ? 'text-red-500' : 'text-slate-400'}`}>
   255→                {formatFileSize(totalSize)} / {formatFileSize(MAX_TOTAL_SIZE)}
   256→              </span>
   257→              <div className="w-20 h-1 rounded-full bg-slate-100 dark:bg-slate-700">
   258→                <div
   259→                  className={`h-full rounded-full transition-all duration-300 ${
   260→                    totalSize > MAX_TOTAL_SIZE ? 'bg-red-500' : totalSize > MAX_TOTAL_SIZE * 0.8 ? 'bg-amber-500' : 'bg-amber-500'
   261→                  }`}
   262→                  style={{ width: `${Math.min((totalSize / MAX_TOTAL_SIZE) * 100, 100)}%` }}
   263→                />
   264→              </div>
   265→            </div>
   266→          </div>
   267→        </div>
   268→      )}
   269→    </div>
   270→  );
   271→}

   273→function useOfficeStatus() {
   274→  const status = useMemo(() => {
   275→    const now = new Date();
   276→    const utcHours = now.getUTCHours();
   277→    const day = now.getUTCDay();
   278→    const ghHour = utcHours;
   279→
   280→    if (day === 0) return { open: false, label: 'Currently Closed' };
   281→    if (day === 6) {
   282→      if (ghHour >= 9 && ghHour < 13) return { open: true, label: 'Currently Open' };
   283→      return { open: false, label: 'Currently Closed' };
   284→    }
   285→    if (ghHour >= 8 && ghHour < 17) return { open: true, label: 'Currently Open' };
   286→    return { open: false, label: 'Currently Closed' };
   287→  }, []);
   288→  return status;
   289→}
